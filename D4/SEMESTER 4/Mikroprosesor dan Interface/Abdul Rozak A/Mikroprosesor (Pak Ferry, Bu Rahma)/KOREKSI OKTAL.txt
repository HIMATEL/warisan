KOR1 PROC
    PUSH SI
    MOV SI,DI
    AND SI,000FH
    CMP SI,8
    JB  AA
    ADD DI,0008H
AA: MOV SI,DI
    AND SI,00F0H
    CMP SI,80H
    JB  BB
    ADD DI,0080H
BB: MOV SI,DI
    AND SI,0F00H
    CMP SI,800H
    JB  CC
    ADD DI,0800H
    ADC AX,0
CC: MOV SI,DI
    AND SI,0F000H
    CMP SI,8000H
    JB  DD
    ADD DI,8000H
    ADC AX,0
DD: MOV SI,AX
    AND SI,000FH
    CMP SI,0008H
    JB  EE
    ADD AX,0008H
EE: POP SI
    
    RET
KOR1 ENDP
