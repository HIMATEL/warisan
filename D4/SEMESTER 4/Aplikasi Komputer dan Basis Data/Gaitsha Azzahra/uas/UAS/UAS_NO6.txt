TITLE UAS NO_6 ;konversi oktal-desimal
.MODEL SMALL
.STACK 100H
.DATA  
TABEL1 DB 0, 1H, 2H, 3H, 4H
       DB 5H, 6H, 7H

TABEL2 DB 0, 8H, 16H, 24H
       DB 32H, 40H, 48H, 56H

.CODE
MAIN PROC
    MOV AX, @DATA
    MOV DS, AX
    
    MOV AL, CH
    AND AL, 0FH
    MOV AH, 0
    MOV SI, AX
    MOV BL, [TABEL1 + SI]
    
    MOV AL, CH
    AND AL, 0F0H
    SHR AL, 4
    MOV AH, 0
    MOV SI, AX
    MOV CL, [TABEL2 + SI]  
    
    MOV AL, CL
    ADD AL, BL
    DAA
    MOV CL, AL
    
    MOV AH, 4CH
    INT 21H
    
MAIN ENDP 
    END MAIN
